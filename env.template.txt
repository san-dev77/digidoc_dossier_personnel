# ===== ENVOI DES CODES MFA PAR E-MAIL (SMTP) =====
# Ce bloc n'est pas vraiment optionnel en production : l'e-mail est le canal
# principal du MFA. Tant que SMTP_HOST est vide, les codes ne sont PAS envoyés,
# ils sont écrits en clair dans les journaux (/logs) — donc seul le compte de
# secours `digidoc` (mfa_exempt) peut se connecter sans qu'un administrateur
# aille lire le journal.
#
# Deux règles qui font échouer la plupart des premières tentatives :
#  - SMTP_FROM doit être une adresse que le relais autorise. Avec un fournisseur
#    comme Gmail, elle doit être IDENTIQUE à SMTP_USER.
#  - Si le relais interne filtre par adresse IP au lieu d'authentifier, laissez
#    SMTP_USER et SMTP_PASSWORD VIDES : le code n'enverra alors aucun
#    identifiant, ce qui est le comportement attendu d'un relais ouvert.
#
# Vérifiez la configuration AVANT de tester une connexion :
#     npm run test:smtp
# Le script teste la connexion, l'authentification, puis envoie un message.
#
# Ces valeurs sont lues au DÉMARRAGE : après modification, redémarrez le
# service (sudo systemctl restart digidoc), sinon l'ancienne config reste active.
SMTP_HOST=""
SMTP_PORT="587"
SMTP_SECURE="false"
SMTP_USER=""
SMTP_PASSWORD=""
SMTP_FROM=""

# Relais interne dont le certificat est signe par une autorite privee
# (erreur "self signed certificate in certificate chain") :
#  - SMTP_CA_FILE : chemin du certificat PEM de l'autorite. Recommande, la
#    verification du certificat reste active.
#  - SMTP_TLS_INSECURE=true : desactive la verification. A reserver a un
#    reseau interne de confiance, quand le PEM est introuvable.
# SMTP_CA_FILE="/etc/ssl/certs/ca-interne.pem"
# SMTP_TLS_INSECURE="false"
